19.1 引子：从 ch18 研究助手到 ch19 数据分析

ch18 把"信源可追溯"带给了研究场景——一篇论文里的每个论点都能追到 arxiv:XXXX.YYYYY。ch19 要把"信源可追溯"扩展到数据场景——一张报表里的每个数字都能追到 SELECT … FROM … 的 SQL 字符串与底层的源表。这不是"再加几个工具"，是新增一种血缘追踪能力。

19.1.5 工程意义：数据分析 Agent 在生产中的 3 个真实场景

1. BI 报表自动化：每天拉取昨日销售数据 → 出 5 张报表 → 邮件给高管。ch19 主用例。 2. 自助式 SQL：产品经理用自然语言问"上周哪个渠道转化率最高" → Agent 自动写 SQL + 出图。 3. 数据科学协作：数据分析师让 Agent 做特征工程（聚合 / 窗口函数）→ 自己专注模型调优。

19.3 30 行 Hello World：最小 Supervisor + SQLAgent

我们先实现最小骨架：1 个 Supervisor + 1 个 SQLAgent Worker，能把用户业务问题拆成 SQL、查询 DuckDB、返回结果。

19.3.1 supervisor.py + sql_agent.py 第 1 版（30 行）

# ch19 / supervisor.py — 最小骨架（30 行）
# 注 1：以下代码为 mock 模式（call_llm 返回固定 SQL），无需 API key；
#       切真实 LLM 见 §19.9.4a；切真实 DuckDB 见 §19.9.4b
# 注 2：§19.8.3 SQL 注入示例需 DuckDB（pip install duckdb）才能跑——本节 mock 模式不要混淆
import json, re

SUPER_PROMPT = """你是数据分析 Supervisor。任务：把业务问题拆成 1-3 个 SQL 子问题。
业务问题：{question}
输出 JSON：{{"sub_questions": ["...", "..."]}}"""

SQL_PROMPT = """你是 SQLAgent。任务：根据子问题写 1 条 SELECT 语句。
子问题：{sub_q}
输出 JSON：{{"sql": "SELECT ...", "tables": ["..."]}}"""

def call_llm(prompt: str) -> str:
    if "Supervisor" in prompt:
        q = re.search(r"业务问题：(.+)", prompt).group(1)
        return json.dumps({"sub_questions": [f"子问题1：{q}", f"子问题2：{q} 同比"]}, ensure_ascii=False)
    sq = re.search(r"子问题：(.+)", prompt).group(1)
    return json.dumps({"sql": f"SELECT * FROM t WHERE topic = '{sq}' LIMIT 10",
                       "tables": ["t"]}, ensure_ascii=False)

def run(question: str) -> dict:
    plan = json.loads(call_llm(SUPER_PROMPT.format(question=question)))
    sqls = [json.loads(call_llm(SQL_PROMPT.format(sub_q=sq))) for sq in plan["sub_questions"]]
    return {"plan": plan, "sqls": sqls}

if __name__ == "__main__":
    print(json.dumps(run("上周销售"), ensure_ascii=False, indent=2))

19.4.3 tools.py handlers（5 关键 handler）

# ch19 / tools.py — 关键 5 个 handler（节选）
# 注：完整版含 12 handler，本节只列 ch19 新增/修改

ALLOWED_TABLES = {"sales", "users", "products", "events"}  # 第 5 层防护：白名单
MAX_ROWS = 1000  # 第 5 层防护：行数上限

def sql_execute(sql: str, dry_run: bool = True, max_rows: int = MAX_ROWS) -> dict:
    """只读 SELECT，带白名单 + LIMIT 防护"""
    sql_clean = sql.strip().rstrip(";")
    # L5 防护 1：白名单表名
    tables = _extract_tables(sql_clean)
    if not tables.issubset(ALLOWED_TABLES):
        return {"error": f"table not in whitelist: {tables - ALLOWED_TABLES}"}
    # L5 防护 2：禁止 DDL/DML
    if not sql_clean.upper().startswith("SELECT") and not sql_clean.upper().startswith("WITH"):
        return {"error": "only SELECT/WITH allowed"}
    # L5 防护 3：强制 LIMIT
    if "LIMIT" not in sql_clean.upper():
        sql_clean += f" LIMIT {max_rows}"
    if dry_run:
        return {"dry_run": True, "sql": sql_clean, "would_return": "<EXPLAIN>"}
    # 真实执行（mock 返回）
    return {"rows": [{"topic": t, "cnt": 42} for t in tables], "sql": sql_clean}

def chart_plot(data: list, chart_type: str = "bar", title: str = "") -> dict:
    """出图（mock 模式返回 SVG 字符串）"""
    if chart_type not in {"bar", "line", "pie", "scatter"}:
        return {"error": f"chart_type {chart_type} not supported"}
    svg = f'<svg width="400" height="300"><text x="200" y="150">{title}</text></svg>'
    return {"chart_type": chart_type, "svg": svg, "rows_used": len(data)}

def schema_list() -> dict:
    """返回白名单表的 schema（SQLAgent 必须先看 schema 再写 SQL）"""
    return {"tables": {t: ["id", "topic", "cnt", "ts"] for t in ALLOWED_TABLES}}

def _extract_tables(sql: str) -> set:
    """极简表名提取（生产应使用 sqlglot）"""
    import re
    return set(re.findall(r'(?:FROM|JOIN)\s+(\w+)', sql, re.IGNORECASE))

关键点：

1. L5 三件套：白名单 + DDL/DML 禁止 + LIMIT 强制 2. dry_run=true 默认：先 EXPLAIN 不执行，防止误删 3. schema_list 在 sql_execute 之前：SQLAgent 必须先看 schema，再写 SQL

19.5.6 Supervisor 类继承（从 ch18）

# ch19 / supervisor.py — 继承 ch18 Supervisor
from supervisor_ch18 import Supervisor as SupervisorCh18

class Supervisor(SupervisorCh18):
    """ch19 Supervisor：继承 ch18 + 加 SQLAgent/ChartAgent 委派"""
    def __init__(self):
        super().__init__()
        self.workers = {
            "sql": SQLAgent(),
            "chart": ChartAgent(),
            "research": Researcher(),  # ch18 复用
            "write": Writer(),
            "review": Reviewer(),
        }

    def delegate(self, task: str, worker: str) -> dict:
        """委派给指定 Worker（ch18 风格的统一入口）"""
        if worker not in self.workers:
            return {"error": f"unknown worker: {worker}"}
        return self.workers[worker].run(task)

class LineageGraph:
    """血缘三元组：(query_sql, source_table, chart_id)"""
    def __init__(self):
        self.edges = []  # list of (sql, table, chart_id, ts)

    def add(self, sql: str, table: str, chart_id: str):
        self.edges.append((sql, table, chart_id, time.time()))

    def trace_chart(self, chart_id: str) -> list:
        """反向追溯：这张图来自哪条 SQL → 哪张表"""
        return [(s, t) for s, t, c, _ in self.edges if c == chart_id]

    def trace_table(self, table: str) -> list:
        """正向追溯：这张表被哪些图用到"""
        return [c for s, t, c, _ in self.edges if t == table]

    def detect_conflict(self) -> list:
        """冲突检测：同一表的不同 SQL 给出冲突数字"""

        return []

# ch19 / supervisor.py — 5 层防护（节选）
def safety_check(task: dict, worker: str) -> bool:
    """5 层防护统一入口"""
    # L1: 工具白名单
    if task.get("tool") not in ALLOWED_TOOLS:
        return False
    # L2: 委派 token（Worker 身份）
    if task.get("token") not in ACTIVE_TOKENS.get(worker, []):
        return False
    # L3: 任务范围限制
    if not check_scope(worker, task):
        return False
    # L4: Trace log（已自动记录在 decorator）
    # L5: SQL 注入防护
    if task.get("tool") == "sql_execute":
        return sql_l5_check(task.get("args", {}).get("sql", ""))
    return True

def sql_l5_check(sql: str) -> bool:
    """L5 防护：白名单表 + 禁止 DDL/DML + LIMIT 强制"""
    if not sql.upper().startswith(("SELECT", "WITH")):
        return False
    tables = _extract_tables(sql)
    if not tables.issubset(ALLOWED_TABLES):
        return False
    if "LIMIT" not in sql.upper():
        return False  # 强制带 LIMIT
    return True

19.8.3 SQL 注入示例：参数化 vs 字符串拼接

# === 反例：字符串拼接（SQL 注入）===
